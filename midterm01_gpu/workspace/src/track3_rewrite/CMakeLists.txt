cmake_minimum_required(VERSION 3.18)
project(track3_rewrite LANGUAGES C CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io sample_consensus segmentation)
find_package(CUDAToolkit REQUIRED)
find_package(OpenSSL REQUIRED)
add_library(track3_geometry src/geometry.cu src/registration.cpp)
target_include_directories(track3_geometry PUBLIC include)
target_link_libraries(track3_geometry PUBLIC Eigen3::Eigen CUDA::cudart)
target_compile_options(track3_geometry PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--expt-relaxed-constexpr>)
add_executable(track3_rewrite src/main.cpp)
target_include_directories(track3_rewrite PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(track3_rewrite PRIVATE track3_geometry ${PCL_LIBRARIES} OpenSSL::Crypto)
add_executable(track3_mechanisms tests/mechanisms.cpp)
target_link_libraries(track3_mechanisms PRIVATE track3_geometry)
enable_testing()
add_test(NAME native_gpu_mechanisms COMMAND track3_mechanisms)
